package main

import (
	"context"
	"fmt"
	"log"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"project-ticketing-golang/internal/config"
	"project-ticketing-golang/internal/handler"
	"project-ticketing-golang/internal/middleware"
	"project-ticketing-golang/internal/repository"
	"project-ticketing-golang/internal/service"
)

func main() {
	cfg := config.LoadConfig()

	db, err := repository.NewPostgresDB(cfg.DBHost, cfg.DBPort, cfg.DBUser, cfg.DBPassword, cfg.DBName, cfg.DBSSLMode)
	if err != nil {
		log.Fatalf("Gagal koneksi ke database: %v", err)
	}
	defer db.Close()
	log.Println("Berhasil terhubung ke database")

	repository.SeedAdmin(context.Background(), db, cfg.AdminEmail, cfg.AdminPassword)

	userRepo := repository.NewUserRepository(db)
	eventRepo := repository.NewEventRepository(db)
	orderRepo := repository.NewOrderRepository(db)

	authService := service.NewAuthService(userRepo, cfg)
	eventService := service.NewEventService(eventRepo)
	orderService := service.NewOrderService(orderRepo)

	authHandler := handler.NewAuthHandler(authService)
	eventHandler := handler.NewEventHandler(eventService)
	orderHandler := handler.NewOrderHandler(orderService)

	mux := http.NewServeMux()

	// Public Routes
	mux.HandleFunc("GET /health", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusOK)
		w.Write([]byte(`{"success": true, "message": "Service sehat", "data": {"status": "up", "database": "up"}}`))
	})
	mux.HandleFunc("POST /api/v1/auth/register", authHandler.Register)
	mux.HandleFunc("POST /api/v1/auth/login", authHandler.Login)
	mux.HandleFunc("GET /api/v1/events", eventHandler.ListPublic)
	mux.HandleFunc("GET /api/v1/events/{id}", eventHandler.GetDetail)

	// Auth Routes (Customer/Admin)
	authMux := http.NewServeMux()
	authMux.HandleFunc("GET /api/v1/auth/me", authHandler.Me)
	mux.Handle("/api/v1/auth/me", middleware.AuthMiddleware([]byte(cfg.JWTSecret), "")(authMux))

	// Customer Only Routes
	customerMux := http.NewServeMux()
	customerMux.HandleFunc("POST /api/v1/orders", orderHandler.Create)
	customerMux.HandleFunc("POST /api/v1/orders/{id}/pay", orderHandler.Pay)
	customerMux.HandleFunc("POST /api/v1/orders/{id}/cancel", orderHandler.Cancel)
	mux.Handle("/api/v1/orders", middleware.AuthMiddleware([]byte(cfg.JWTSecret), "customer")(customerMux))
	mux.Handle("/api/v1/orders/", middleware.AuthMiddleware([]byte(cfg.JWTSecret), "customer")(customerMux))

	// Admin Only Routes
	adminMux := http.NewServeMux()
	adminMux.HandleFunc("POST /api/v1/events", eventHandler.Create)
	adminMux.HandleFunc("PUT /api/v1/events/{id}", eventHandler.Update)
	adminMux.HandleFunc("PATCH /api/v1/events/{id}/publish", eventHandler.Publish)
	adminMux.HandleFunc("PATCH /api/v1/events/{id}/cancel", eventHandler.Cancel)
	mux.Handle("/api/v1/events", middleware.AuthMiddleware([]byte(cfg.JWTSecret), "admin")(adminMux))
	mux.Handle("/api/v1/events/", middleware.AuthMiddleware([]byte(cfg.JWTSecret), "admin")(adminMux))

	server := &http.Server{
		Addr:    ":" + cfg.AppPort,
		Handler: mux,
	}

	go func() {
		log.Printf("Server berjalan di port %s", cfg.AppPort)
		if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Fatalf("Server error: %v", err)
		}
	}()

	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	<-quit
	log.Println("Mematikan server...")

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	if err := server.Shutdown(ctx); err != nil {
		log.Fatalf("Server forced to shutdown: %v", err)
	}

	fmt.Println("Server keluar dengan rapi")
}

package config

import (
	"log"
	"os"

	"github.com/joho/godotenv"

type Config struct {
	AppPort       string
	DBHost        string
	DBPort        string
	DBUser        string
	DBPassword    string
	DBName        string
	DBSSLMode     string
	JWTSecret     string
	AdminEmail    string
	AdminPassword string
	WorkerCount   string
	QueueCapacity string
}

func LoadConfig() *Config {
	err := godotenv.Load()
	if err != nil {
		log.Println("Peringatan: File .env tidak ditemukan, menggunakan environment variable sistem")
	}

	return &Config{
		AppPort:       getEnv("APP_PORT", "8080"),
		DBHost:        getEnv("DB_HOST", "localhost"),
		DBPort:        getEnv("DB_PORT", "5442"),
		DBUser:        getEnv("DB_USER", "kiko"),
		DBPassword:    getEnv("DB_PASSWORD", ""),
		DBName:        getEnv("DB_NAME", "ticketing_golang"),
		DBSSLMode:     getEnv("DB_SSLMODE", "disable"),
		JWTSecret:     getEnv("JWT_SECRET", "secret"),
		AdminEmail:    getEnv("ADMIN_EMAIL", "admin@example.com"),
		AdminPassword: getEnv("ADMIN_PASSWORD", "Admin12345"),
		WorkerCount:   getEnv("IMPORT_WORKER_COUNT", "4"),
		QueueCapacity: getEnv("IMPORT_QUEUE_CAPACITY", "10"),
	}
}

func getEnv(key, fallback string) string {
	if value, exists := os.LookupEnv(key); exists {
		return value
	}
	return fallback
}

package handler

import (
	"encoding/json"
	"errors"
	"net/http"

	"project-ticketing-golang/internal/middleware"
	"project-ticketing-golang/internal/model"
	"project-ticketing-golang/internal/service"
)

type AuthHandler struct {
	authService *service.AuthService
}

func NewAuthHandler(authService *service.AuthService) *AuthHandler {
	return &AuthHandler{authService: authService}
}

func (h *AuthHandler) Register(w http.ResponseWriter, r *http.Request) {
	var req model.RegisterRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", "Input tidak valid", nil)
		return
	}

	user, err := h.authService.Register(r.Context(), req)
	if err != nil {
		if errors.Is(err, model.ErrConflict) {
			writeError(w, http.StatusConflict, "CONFLICT", "Email sudah terdaftar", nil)
			return
		}
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", err.Error(), nil)
		return
	}

	writeSuccess(w, http.StatusCreated, "Registrasi berhasil", user, nil)
}

func (h *AuthHandler) Login(w http.ResponseWriter, r *http.Request) {
	var req model.LoginRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", "Input tidak valid", nil)
		return
	}

	token, user, err := h.authService.Login(r.Context(), req)
	if err != nil {
		writeError(w, http.StatusUnauthorized, "UNAUTHORIZED", "Email atau password salah", nil)
		return
	}

	resData := map[string]interface{}{
		"access_token": token,
		"token_type":   "Bearer",
		"expires_in":   3600,
		"user":         user,
	}
	writeSuccess(w, http.StatusOK, "Login berhasil", resData, nil)
}

func (h *AuthHandler) Me(w http.ResponseWriter, r *http.Request) {
	userID, ok := r.Context().Value(middleware.UserContextKey).(string)
	if !ok {
		writeError(w, http.StatusUnauthorized, "UNAUTHORIZED", "Token tidak valid", nil)
		return
	}
	// Sementara kita return ID-nya dulu, atau kamu bisa query ulang ke DB
	writeSuccess(w, http.StatusOK, "Data user berhasil diambil", map[string]string{"id": userID}, nil)
}

// --- HELPER ENVELOPE RESPONSE ---
func writeSuccess(w http.ResponseWriter, status int, message string, data interface{}, meta interface{}) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(model.SuccessResponse{
		Success: true,
		Message: message,
		Data:    data,
		Meta:    meta,
	})
}

func writeError(w http.ResponseWriter, status int, code string, message string, details []model.ErrorDetail) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	var res model.ErrorResponse
	res.Success = false
	res.Message = message
	res.Error.Code = code
	res.Error.Details = details
	if res.Error.Details == nil {
		res.Error.Details = []model.ErrorDetail{}
	}
	json.NewEncoder(w).Encode(res)
}

import (
	"encoding/json"

	"project-ticketing-golang/internal/model"
	"project-ticketing-golang/internal/service"

type EventHandler struct {
	eventService *service.EventService
}

func NewEventHandler(es *service.EventService) *EventHandler {
	return &EventHandler{eventService: es}
}

func (h *EventHandler) Create(w http.ResponseWriter, r *http.Request) {
	adminID, _ := r.Context().Value(middleware.UserContextKey).(string)
	var req model.CreateEventRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", "Input tidak valid", nil)
		return
	}

	event, err := h.eventService.CreateEvent(r.Context(), adminID, req)
	if err != nil {
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", err.Error(), nil)
		return
	}
	writeSuccess(w, http.StatusCreated, "Event berhasil dibuat", event, nil)
}

func (h *EventHandler) Update(w http.ResponseWriter, r *http.Request) {
	adminID, _ := r.Context().Value(middleware.UserContextKey).(string)
	id := r.PathValue("id")
	var req model.UpdateEventRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", "Input tidak valid", nil)
		return
	}

	event, err := h.eventService.UpdateEvent(r.Context(), adminID, id, req)
	if err != nil {
		if err.Error() == "VERSION_CONFLICT" {
			writeError(w, http.StatusConflict, "VERSION_CONFLICT", "Event sudah diubah oleh pengguna lain, muat ulang data terbaru", nil)
			return
		}
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", err.Error(), nil)
		return
	}
	writeSuccess(w, http.StatusOK, "Event berhasil diperbarui", event, nil)
}

func (h *EventHandler) Publish(w http.ResponseWriter, r *http.Request) {
	adminID, _ := r.Context().Value(middleware.UserContextKey).(string)
	id := r.PathValue("id")

	event, err := h.eventService.PublishEvent(r.Context(), adminID, id)
	if err != nil {
		writeError(w, http.StatusConflict, "INVALID_STATE", err.Error(), nil)
		return
	}
	writeSuccess(w, http.StatusOK, "Event berhasil diterbitkan", event, nil)
}

func (h *EventHandler) Cancel(w http.ResponseWriter, r *http.Request) {
	adminID, _ := r.Context().Value(middleware.UserContextKey).(string)
	id := r.PathValue("id")
	var req model.CancelEventRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", "Input tidak valid", nil)
		return
	}

	event, cancelledOrders, err := h.eventService.CancelEvent(r.Context(), adminID, id, req)
	if err != nil {
		if err.Error() == "INVALID_STATE" {
			writeError(w, http.StatusConflict, "INVALID_STATE", "Event sudah dibatalkan", nil)
			return
		}
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", err.Error(), nil)
		return
	}

	// Custom response untuk cancel sesuai brief
	responseData := map[string]interface{}{
		"event":            event,
		"cancelled_orders": cancelledOrders,
	}
	writeSuccess(w, http.StatusOK, "Event berhasil dibatalkan", responseData, nil)
}

func (h *EventHandler) ListPublic(w http.ResponseWriter, r *http.Request) {
	role := "public"
	// Coba ambil role dari token jika ada (tanpa memaksa login)
	authHeader := r.Header.Get("Authorization")
	if authHeader != "" {
		// Logika parsing token sederhana untuk ambil role saja (bisa disempurnakan)
		// Untuk sementara, kita asumsikan jika ada header, itu customer/admin
		// (Nanti bisa dioptimalkan)
		role = "customer"
	}

	query := r.URL.Query()
	page := 1
	limit := 10
	if p := query.Get("page"); p != "" {
		fmt.Sscanf(p, "%d", &page)
	}
	if l := query.Get("limit"); l != "" {
		fmt.Sscanf(l, "%d", &limit)
	}

	filter := map[string]interface{}{
		"search": query.Get("search"),
		"status": query.Get("status"),
		"sort":   query.Get("sort"),
		"order":  query.Get("order"),
	}

	events, totalItems, err := h.eventService.GetEvents(r.Context(), role, filter, page, limit)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "INTERNAL_ERROR", err.Error(), nil)
		return
	}

	totalPages := (totalItems + limit - 1) / limit
	meta := map[string]interface{}{
		"page":        page,
		"limit":       limit,
		"total_items": totalItems,
		"total_pages": totalPages,
	}

	writeSuccess(w, http.StatusOK, "Daftar event berhasil diambil", events, meta)
}

func (h *EventHandler) GetDetail(w http.ResponseWriter, r *http.Request) {
	role := "public" // Default
	id := r.PathValue("id")

	event, err := h.eventService.GetEventDetail(r.Context(), role, id)
	if err != nil {
		if err == model.ErrNotFound {
			writeError(w, http.StatusNotFound, "NOT_FOUND", "Event tidak ditemukan", nil)
			return
		}
		writeError(w, http.StatusInternalServerError, "INTERNAL_ERROR", err.Error(), nil)
		return
	}
	writeSuccess(w, http.StatusOK, "Detail event berhasil diambil", event, nil)
}

type OrderHandler struct {
	orderService *service.OrderService
}

func NewOrderHandler(os *service.OrderService) *OrderHandler {
	return &OrderHandler{orderService: os}
}

func (h *OrderHandler) Create(w http.ResponseWriter, r *http.Request) {
	userID, _ := r.Context().Value(middleware.UserContextKey).(string)

	idempotencyKey := r.Header.Get("Idempotency-Key")
	if idempotencyKey == "" || len(idempotencyKey) > 100 {
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", "Header Idempotency-Key wajib diisi maksimal 100 karakter", nil)
		return
	}

	var req model.CreateOrderRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "VALIDATION_ERROR", "Input tidak valid", nil)
		return
	}

	order, isReplay, err := h.orderService.CreateOrder(r.Context(), userID, idempotencyKey, req)
	if err != nil {
		switch err.Error() {
		case "IDEMPOTENCY_KEY_REUSED":
			writeError(w, http.StatusUnprocessableEntity, "IDEMPOTENCY_KEY_REUSED", "Idempotency-Key sudah dipakai untuk request yang berbeda", nil)
		case "TICKET_LIMIT_EXCEEDED":
			writeError(w, http.StatusConflict, "TICKET_LIMIT_EXCEEDED", "Total tiket customer untuk event tersebut akan melebihi 5", nil)
		case "INSUFFICIENT_QUOTA":
			writeError(w, http.StatusConflict, "INSUFFICIENT_QUOTA", "Sisa tiket tidak mencukupi", nil)
		case "INVALID_STATE":
			writeError(w, http.StatusConflict, "INVALID_STATE", "Event tidak valid atau sudah dimulai", nil)
		default:
			writeError(w, http.StatusInternalServerError, "INTERNAL_ERROR", err.Error(), nil)
		}
		return
	}

	if isReplay {
		w.Header().Set("Idempotent-Replayed", "true")
	}
	writeSuccess(w, http.StatusCreated, "Order berhasil dibuat, selesaikan pembayaran sebelum batas waktu", order, nil)
}

func (h *OrderHandler) Pay(w http.ResponseWriter, r *http.Request) {
	userID, _ := r.Context().Value(middleware.UserContextKey).(string)
	orderID := r.PathValue("id")

	order, err := h.orderService.PayOrder(r.Context(), userID, orderID)
	if err != nil {
		if err == model.ErrNotFound {
			writeError(w, http.StatusNotFound, "NOT_FOUND", "Order tidak ditemukan", nil)
			return
		}
		writeError(w, http.StatusConflict, "INVALID_STATE", "Order tidak dapat dibayar karena status tidak valid atau sudah expired", nil)
		return
	}
	writeSuccess(w, http.StatusOK, "Pembayaran berhasil", order, nil)
}

func (h *OrderHandler) Cancel(w http.ResponseWriter, r *http.Request) {

		return
	}
	writeSuccess(w, http.StatusOK, "Order berhasil dibatalkan", order, nil)
}

package middleware

import (
	"context"
	"fmt"
	"net/http"
	"strings"

	"github.com/golang-jwt/jwt/v5"
)

type contextKey string

const UserContextKey contextKey = "user"

func AuthMiddleware(jwtSecret []byte, requiredRole string) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			w.Header().Set("Content-Type", "application/json")

			authHeader := r.Header.Get("Authorization")
			if !strings.HasPrefix(authHeader, "Bearer ") {
				w.WriteHeader(http.StatusUnauthorized)
				w.Write([]byte(`{"success": false, "message": "Token tidak valid atau sudah kedaluwarsa", "error": {"code": "UNAUTHORIZED", "details": []}}`))
				return
			}

			tokenString := strings.TrimPrefix(authHeader, "Bearer ")
			token, err := jwt.Parse(tokenString, func(token *jwt.Token) (interface{}, error) {
				if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
					return nil, fmt.Errorf("unexpected signing method")
				}
				return jwtSecret, nil
			})

			if err != nil || !token.Valid {
				w.WriteHeader(http.StatusUnauthorized)
				w.Write([]byte(`{"success": false, "message": "Token tidak valid atau sudah kedaluwarsa", "error": {"code": "UNAUTHORIZED", "details": []}}`))
				return
			}

			claims, ok := token.Claims.(jwt.MapClaims)
			if !ok {
				w.WriteHeader(http.StatusUnauthorized)
				w.Write([]byte(`{"success": false, "message": "Token tidak valid", "error": {"code": "UNAUTHORIZED", "details": []}}`))
				return
			}

			// Cek Role (jika endpoint khusus admin)
			role, _ := claims["role"].(string)
			if requiredRole != "" && role != requiredRole {
				w.WriteHeader(http.StatusForbidden)
				w.Write([]byte(`{"success": false, "message": "Anda tidak memiliki akses ke resource ini", "error": {"code": "FORBIDDEN", "details": []}}`))
				return
			}

			// Simpan user ID ke context
			userID, _ := claims["sub"].(string)
			ctx := context.WithValue(r.Context(), UserContextKey, userID)
			next.ServeHTTP(w, r.WithContext(ctx))
		})
	}
}

package model

import "time"

type Event struct {
	ID          string    `json:"id"`
	Name        string    `json:"name"`
	Description string    `json:"description,omitempty"`
	Venue       string    `json:"venue"`
	StartAt     time.Time `json:"start_at"`
	EndAt       time.Time `json:"end_at"`
	Price       float64   `json:"price"`
	Quota       int       `json:"quota"`
	Available   int       `json:"available"`

func NewPostgresDB(host, port, user, password, dbname, sslmode string) (*sql.DB, error) {
	// Gunakan format URL
	dsn := fmt.Sprintf("postgres://%s:%s@%s:%s/%s?sslmode=%s",
		user, password, host, port, dbname, sslmode)

	db, err := sql.Open("pgx", dsn)
	if err != nil {
		return nil, err
	}

	db.SetMaxOpenConns(25)
	db.SetMaxIdleConns(25)
	db.SetConnMaxLifetime(5 * time.Minute)

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	err = db.PingContext(ctx)
	if err != nil {
		return nil, err
	}

	return db, nil
}

func (r *eventRepository) GetEvents(ctx context.Context, filter map[string]interface{}, page, limit int) ([]*model.Event, int, error) {
	baseQuery := `FROM events WHERE 1=1`
	var args []interface{}
	argId := 1

	if status, ok := filter["status"].(string); ok && status != "" {
		baseQuery += fmt.Sprintf(" AND status = $%d", argId)
		args = append(args, status)
		argId++
	}
	if search, ok := filter["search"].(string); ok && search != "" {
		baseQuery += fmt.Sprintf(" AND (LOWER(name) LIKE $%d OR LOWER(venue) LIKE $%d)", argId, argId)
		args = append(args, "%"+strings.ToLower(search)+"%")
		argId++
	}

	// Count total
	var totalItems int
	countQuery := "SELECT COUNT(*) " + baseQuery
	err := r.db.QueryRowContext(ctx, countQuery, args...).Scan(&totalItems)
	if err != nil {
		return nil, 0, err
	}

	// Pagination & Sort
	sortBy := "start_at"
	if s, ok := filter["sort"].(string); ok && s != "" {
		sortBy = s
	}
	order := "ASC"
	if o, ok := filter["order"].(string); ok && strings.ToLower(o) == "desc" {
		order = "DESC"
	}

	offset := (page - 1) * limit
	query := fmt.Sprintf(`SELECT id, name, COALESCE(description, ''), venue, start_at, end_at, price, quota, available, status, version, created_by, created_at, updated_at %s ORDER BY %s %s LIMIT $%d OFFSET $%d`, baseQuery, sortBy, order, argId, argId+1)
	args = append(args, limit, offset)

	rows, err := r.db.QueryContext(ctx, query, args...)
	if err != nil {
		return nil, 0, err
	}
	defer rows.Close()

	var events []*model.Event
	for rows.Next() {
		var e model.Event
		err := rows.Scan(&e.ID, &e.Name, &e.Description, &e.Venue, &e.StartAt, &e.EndAt, &e.Price, &e.Quota, &e.Available, &e.Status, &e.Version, &e.CreatedBy, &e.CreatedAt, &e.UpdatedAt)
		if err != nil {
			return nil, 0, err
		}
		events = append(events, &e)
	}
	return events, totalItems, nil
}

package service

import (
	"context"
	"errors"
	"strings"
	"time"

	"project-ticketing-golang/internal/config"
	"project-ticketing-golang/internal/model"
	"project-ticketing-golang/internal/repository"

	"github.com/golang-jwt/jwt/v5"
	"golang.org/x/crypto/bcrypt"
)

type AuthService struct {
	userRepo  repository.UserRepository
	jwtSecret []byte
}

func NewAuthService(userRepo repository.UserRepository, cfg *config.Config) *AuthService {
	return &AuthService{
		userRepo:  userRepo,
		jwtSecret: []byte(cfg.JWTSecret),
	}
}

func (s *AuthService) Register(ctx context.Context, req model.RegisterRequest) (*model.User, error) {
	// Validasi sederhana
	if len(req.Password) < 8 || len(req.Password) > 72 {
		return nil, errors.New("VALIDATION_ERROR:password harus 8-72 karakter")
	}

	// Cek email sudah ada atau belum
	existing, _ := s.userRepo.FindByEmail(ctx, req.Email)
	if existing != nil {
		return nil, model.ErrConflict
	}

	hashedPassword, err := bcrypt.GenerateFromPassword([]byte(req.Password), bcrypt.DefaultCost)
	if err != nil {
		return nil, err
	}

	user := &model.User{
		Name:         req.Name,
		Email:        strings.ToLower(req.Email),
		PasswordHash: string(hashedPassword),
		Role:         "customer", // Sesuai brief: register selalu customer
	}

	err = s.userRepo.Create(ctx, user)
	if err != nil {
		return nil, err
	}

	return user, nil
}

func (s *AuthService) Login(ctx context.Context, req model.LoginRequest) (string, *model.User, error) {
	user, err := s.userRepo.FindByEmail(ctx, req.Email)
	if err != nil {
		return "", nil, model.ErrUnauthorized // Pesan error disamakan sesuai brief
	}

	err = bcrypt.CompareHashAndPassword([]byte(user.PasswordHash), []byte(req.Password))
	if err != nil {
		return "", nil, model.ErrUnauthorized
	}

	// Buat JWT
	claims := jwt.MapClaims{
		"sub":  user.ID,
		"role": user.Role,
		"iat":  time.Now().Unix(),
		"exp":  time.Now().Add(time.Hour).Unix(), // Expire 1 jam
	}

	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
	tokenString, err := token.SignedString(s.jwtSecret)
	if err != nil {
		return "", nil, err
	}

	return tokenString, user, nil
}

func (s *EventService) CreateEvent(ctx context.Context, adminID string, req model.CreateEventRequest) (*model.Event, error) {
	startAt, _ := time.Parse(time.RFC3339, req.StartAt)
	endAt, _ := time.Parse(time.RFC3339, req.EndAt)

	if startAt.Before(time.Now()) {
		return nil, errors.New("start_at harus di masa depan")
	}
	if !endAt.After(startAt) {
		return nil, errors.New("end_at harus setelah start_at")
	}

	event := &model.Event{
		Name:        req.Name,
		Description: req.Description,
		Venue:       req.Venue,
		StartAt:     startAt,
		EndAt:       endAt,
		Price:       req.Price,
		Quota:       req.Quota,
		Available:   req.Quota,
		CreatedBy:   adminID,
	}

	tx, err := s.eventRepo.BeginTx(ctx)
	if err != nil {
		return nil, err
	}
	defer tx.Rollback()

	err = s.eventRepo.Create(ctx, tx, event)
	if err != nil {
		return nil, err
	}

	afterJSON, _ := json.Marshal(event)
	audit := &model.AuditLog{
		ActorID:    &adminID,
		ActorType:  "user",
		EntityType: "event",
		EntityID:   event.ID,
		Action:     "create",
		Before:     nil,
		After:      afterJSON,
	}

		return nil, err
	}

func (s *EventService) UpdateEvent(ctx context.Context, adminID, id string, req model.UpdateEventRequest) (*model.Event, error) {
	tx, err := s.eventRepo.BeginTx(ctx)
	if err != nil {
		return nil, err
	}
	defer tx.Rollback()

	event, err := s.eventRepo.FindByIDForUpdate(ctx, tx, id)
	if err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return nil, model.ErrNotFound
		}
		return nil, err
	}

	if event.Status == "cancelled" {
		return nil, errors.New("event cancelled tidak bisa diubah")
	}

	if event.Version != req.Version {
		return nil, errors.New("VERSION_CONFLICT")
	}

	beforeJSON, _ := json.Marshal(event)

	startAt, _ := time.Parse(time.RFC3339, req.StartAt)
	endAt, _ := time.Parse(time.RFC3339, req.EndAt)

	terpakai := event.Quota - event.Available
	if req.Quota < terpakai {
		return nil, errors.New("quota tidak boleh lebih kecil dari tiket terpakai")
	}

	event.Name = req.Name
	event.Description = req.Description
	event.Venue = req.Venue
	event.StartAt = startAt
	event.EndAt = endAt
	event.Price = req.Price
	event.Quota = req.Quota
	event.Available = req.Quota - terpakai

	err = s.eventRepo.Update(ctx, tx, event)
	if err != nil {
		return nil, err
	}

	event.Version++
	event.UpdatedAt = time.Now()

	afterJSON, _ := json.Marshal(event)
	audit := &model.AuditLog{
		ActorID:    &adminID,
		ActorType:  "user",
		EntityType: "event",
		EntityID:   event.ID,
		Action:     "update",
		Before:     beforeJSON,
		After:      afterJSON,
	}
	s.eventRepo.InsertAuditLog(ctx, tx, audit)

	if err := tx.Commit(); err != nil {
		return nil, err
	}
	return event, nil
}

func (s *EventService) CancelEvent(ctx context.Context, adminID, id string, req model.CancelEventRequest) (*model.Event, int64, error) {

		return nil, 0, errors.New("reason wajib, 5-255 karakter")
	}

		return nil, 0, err
	}
	defer tx.Rollback()

	event, err := s.eventRepo.FindByIDForUpdate(ctx, tx, id)
	if err != nil {
		return nil, 0, model.ErrNotFound
	}

	if event.Status == "cancelled" {
		return nil, 0, errors.New("INVALID_STATE")
	}

	beforeJSON, _ := json.Marshal(event)

	// Ubah status event
	s.eventRepo.UpdateStatus(ctx, tx, id, "cancelled")
	event.Status = "cancelled"
	event.Version++

	// Batalkan semua order pending & paid
	cancelledOrders, err := s.eventRepo.CancelPendingPaidOrders(ctx, tx, id)
	if err != nil {
		return nil, 0, err
	}

	afterJSON, _ := json.Marshal(event)

	}
	s.eventRepo.InsertAuditLog(ctx, tx, audit)

		return nil, 0, err
	}

package service

import (
	"context"
	"crypto/sha256"
	"database/sql"
	"encoding/hex"
	"encoding/json"
	"errors"
	"time"

	"project-ticketing-golang/internal/model"
	"project-ticketing-golang/internal/repository"
)

type OrderService struct {
	orderRepo repository.OrderRepository
}

func NewOrderService(orderRepo repository.OrderRepository) *OrderService {
	return &OrderService{orderRepo: orderRepo}
}

func (s *OrderService) CreateOrder(ctx context.Context, userID, idempotencyKey string, req model.CreateOrderRequest) (*model.Order, bool, error) {
	// 1. Validasi input
	if req.Quantity < 1 || req.Quantity > 5 {
		return nil, false, errors.New("quantity harus antara 1-5")
	}

	// Hitung hash body request untuk idempotency
	bodyBytes, _ := json.Marshal(req)
	hash := sha256.Sum256(bodyBytes)
	reqHash := hex.EncodeToString(hash[:])

	tx, err := s.orderRepo.BeginTx(ctx)
	if err != nil {
		return nil, false, err
	}
	defer tx.Rollback()

	// 2. Cek Idempotency (SELECT ... FOR UPDATE)
	existingKey, err := s.orderRepo.GetIdempotencyKey(ctx, tx, userID, idempotencyKey)
	if err == nil {
		// Key sudah ada
		if existingKey.RequestHash != reqHash {
			return nil, false, errors.New("IDEMPOTENCY_KEY_REUSED")
		}
		// Body sama, kembalikan response sebelumnya
		var prevOrder model.Order
		json.Unmarshal(existingKey.ResponseBody, &prevOrder)
		return &prevOrder, true, nil
	} else if !errors.Is(err, sql.ErrNoRows) {
		return nil, false, err
	}

	// 3. Lock Event Row & Cek Status/Kuota
	event, err := s.orderRepo.GetEventForUpdate(ctx, tx, req.EventID)
	if err != nil {
		return nil, false, model.ErrNotFound
	}

	if event.Status != "published" {
		return nil, false, errors.New("INVALID_STATE")
	}
	if event.StartAt.Before(time.Now()) {
		return nil, false, errors.New("INVALID_STATE")
	}

	// 4. Cek Batas 5 Tiket per Customer
	existingTickets, err := s.orderRepo.CheckUserTicketLimit(ctx, tx, userID, req.EventID)
	if err != nil {
		return nil, false, err
	}
	if existingTickets+req.Quantity > 5 {
		return nil, false, errors.New("TICKET_LIMIT_EXCEEDED")
	}

	// 5. Kurangi Kuota (Atomic Update)
	err = s.orderRepo.ReduceEventQuota(ctx, tx, req.EventID, req.Quantity)
	if err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return nil, false, errors.New("INSUFFICIENT_QUOTA")
		}
		return nil, false, err
	}

	// 6. Buat Order
	order := &model.Order{
		UserID:     userID,
		EventID:    req.EventID,
		Quantity:   req.Quantity,
		UnitPrice:  event.Price,
		TotalPrice: float64(req.Quantity) * event.Price,
		Status:     "pending",
		ExpiresAt:  time.Now().Add(15 * time.Minute),
	}

	err = s.orderRepo.CreateOrder(ctx, tx, order)
	if err != nil {
		return nil, false, err
	}

	// Siapkan response object
	order.Event = &model.EventInfo{ID: event.ID, Name: event.Name, StartAt: event.StartAt}

	// 7. Simpan Idempotency Key (Hanya jika sukses)
	responseBody, _ := json.Marshal(order)
	ik := &model.IdempotencyKey{
		Key:            idempotencyKey,
		UserID:         userID,
		RequestHash:    reqHash,
		ResponseStatus: 201,
		ResponseBody:   responseBody,
	}
	err = s.orderRepo.SaveIdempotencyKey(ctx, tx, ik)
	if err != nil {
		return nil, false, err
	}

	if err := tx.Commit(); err != nil {
		return nil, false, err
	}

	return order, false, nil
}

func (s *OrderService) PayOrder(ctx context.Context, userID, orderID string) (*model.Order, error) {
	tx, err := s.orderRepo.BeginTx(ctx)
	if err != nil {
		return nil, err
	}
	defer tx.Rollback()

	order, err := s.orderRepo.FindOrderByIDForUpdate(ctx, tx, orderID)
	if err != nil {
		return nil, model.ErrNotFound
	}

	// Cek kepemilikan (jika bukan miliknya, return 404)
	if order.UserID != userID {
		return nil, model.ErrNotFound
	}

	if order.Status != "pending" {
		return nil, errors.New("INVALID_STATE")
	}
	if order.ExpiresAt.Before(time.Now()) {
		return nil, errors.New("INVALID_STATE")
	}

	s.orderRepo.UpdateOrderStatus(ctx, tx, orderID, "paid")
	if err := tx.Commit(); err != nil {
		return nil, err
	}

	return s.orderRepo.FindOrderByID(ctx, orderID)
}

func (s *OrderService) CancelOrder(ctx context.Context, userID, orderID string) (*model.Order, error) {
	tx, err := s.orderRepo.BeginTx(ctx)
	if err != nil {
		return nil, err
	}
	defer tx.Rollback()

	order, err := s.orderRepo.FindOrderByIDForUpdate(ctx, tx, orderID)
	if err != nil {
		return nil, model.ErrNotFound
	}

	if order.UserID != userID {
		return nil, model.ErrNotFound
	}

	if order.Status != "pending" && order.Status != "paid" {
		return nil, errors.New("INVALID_STATE")
	}

	// Cek apakah event sudah mulai
	event, err := s.orderRepo.GetEventForUpdate(ctx, tx, order.EventID)
	if err != nil {
		return nil, err
	}
	if event.StartAt.Before(time.Now()) {
		return nil, errors.New("INVALID_STATE")
	}

	s.orderRepo.UpdateOrderStatus(ctx, tx, orderID, "cancelled")
	s.orderRepo.ReturnEventQuota(ctx, tx, order.EventID, order.Quantity)

	if err := tx.Commit(); err != nil {
		return nil, err
	}

	return s.orderRepo.FindOrderByID(ctx, orderID)
}
